--FULL BACKUP
BACKUP DATABASE dbname TO DISK = 'C:\path\filename.BAK'

--FILE LEVEL BACKUP
BACKUP DATABASE dbname FILE = 'filename' TO DISK = 'C:\filename_backup.FIL'

--FILE GROUP BACKUP
BACKUP DATABASE dbname FILEGROUP = 'filegroupname' TO DISK = 'C:\filegroup_backup.FLG'

--FULL BACKUP MULTIPLE FILES
BACKUP DATABASE dbname TO DISK = 'C:\dbname_1.BAK', DISK = 'D:\dbname_2.BAK', DISK='E:\dbname_3.BAK'

--FULL BACKUP WITH PASSWORD
BACKUP DATABASE dbname TO DISK = 'C:\dbname.BAK' WITH PASSWORD='ASDFGHJKL'

--FULL BACKUP WITH STATS
BACKUP DATABASE dbname TO DISK = 'C:\dbname.BAK' WITH STATS

--FULL BACKUP WITH DESCRIPTION
BACKUP DATABASE dbname TO DISK = 'C:\dbname.BAK' WITH DESCRIPTION = 'description content'

--FULL BACKUP WITH MIRROR
BACKUP DATABASE dbname TO DISK = 'C:\dbname.BAK' MIRROR TO DISK = 'D:\dbname_mirror.BAK' WITH FORMAT
